### train ex1 multi
```bash
  export DIFFSYNTH_MODEL_BASE_PATH=/data-training/models
  export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

  accelerate launch \
    --config_file examples/wanvideo/model_training/full/accelerate_config_zero3.yaml \
    examples/wanvideo/model_training/train.py \
    --task sft:train \
    --dataset_base_path /data-training/multi_face/cache_s2v_f81_fps16_multi \
    --require_cache_manifest \
    --model_id_with_origin_paths "Wan-AI/Wan2.2-S2V-14B:diffusion_pytorch_model*.safetensors" \
    --lora_base_model dit \
    --lora_target_modules "q,k,v,o,ffn.0,ffn.2" \
    --lora_rank 32 \
    --learning_rate 1e-4 \
    --num_epochs 100 \
    --save_steps 50 \
    --remove_prefix_in_ckpt "pipe.dit." \
    --output_path ./models/train/Wan2.2-S2V-14B_lora_multi_zero3 \
    --s2v_ref_rope_mode source_id_local \
    --resume_from_checkpoint "/data-training/hjq/a800_step-9000_sa_rope.safetensors" \
    --use_gradient_checkpointing_offload
```
